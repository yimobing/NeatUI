 1、封装模板 

            
 前端独立 JS 插件最通用、最基础的标准化封装模板（模板一、模板二） 

        
*******************************************************************************************


        下面两套封闭模板是传统前端 JS 插件最基础、标准的 UMD 封装模板，兼顾模块化与浏览器直接使用；
        开发纯原生工具插件，推荐使用模板一，减少外部依赖限制。

        
---------------------------------------------------------

            
 模板一：无依赖原生 JS 插件模板（纯原生，不依赖任何第三方库） 

        
---------------------------------------------------------

        适用场景：纯原生工具类、日期处理、工具函数、独立组件，无需 jQuery/Underscore 等外部库

        
 · 极简注释版 

        

            ;(function (root, factory) {
                if (typeof define === 'function' && define.amd) { // AMD规范
                    define(factory);
                } 
                else if (typeof exports === 'object' && typeof module !== 'undefined') { // UMD规范：Node, CommonJS规范（标准环境）
                    module.exports = factory();
                } 
                else if (typeof exports === 'object') { // UMD规范：Node, CommonJS规范（兜底环境）
                    exports.NeExample = factory();
                } 
                else { // 浏览器全局变量(root 即 window)
                    root.NeExample = factory(); // 全局环境下，挂载到 root（window）上
                }
            })(this, function () {
                // MyPlugin 是临时变量，可以是任何名称，也可命名为跟 root.NeExample同名的 NeExample 
                var MyPlugin = {
                    init: function () {
                        console.log('初始化');
                    }
                };
                return MyPlugin;
            });

 · 全注释版 

        

            ;(function (root, factory) {
                // ①.AMD规范：RequireJS 模块化（无依赖，无需传依赖数组）
                if (typeof define === 'function' && define.amd) {
                    define(factory);
                }
                // ②.UMD规范：CommonJS规范：Node、打包工具(Webpack、Vite)、浏览器ESM环境(如原生type="module")
                // 标准环境：同时存在module与exports
                else if (typeof exports === 'object' && typeof module !== 'undefined') {
                    module.exports = factory();
                }
                // 兜底环境：仅存在exports、无module对象的轻量（极少场景）。老式极简 CommonJS 模拟环境、非常老旧的打包工具才会出现，日常开发几乎碰不到。
                else if (typeof exports === 'object') {
                    exports.NeExample = factory(); // 将插件挂载到exports对象上
                }
                // ③.普通浏览器原生环境：挂载全局window
                else {
                    root.NeExample = factory(); // root = this，外层this指向window，挂载全局变量直接调用
                }
            })(this, function () {
                var MyPlugin = {
                    init: function () {
                        console.log('初始化');
                    }
                };
                return MyPlugin;
            });

        


        模板使用注意事项：
        （1）HTML 页面直接 script 引入即可，无需提前加载任何第三方库；
        （2）RequireJS、Node 环境均不会强制加载额外依赖，无缺失报错；
        （3）内部仅可使用原生 DOM、原生 JS 语法，不可使用$、_.xxx()。




        
---------------------------------------------------------

            
 模板二：带外部依赖插件模板（依赖 jQuery、Underscore） 

        
---------------------------------------------------------

        适用场景：插件需要依赖第三方库（JQ、工具库 Underscore/Lodash 等）

        

            ;(function (root, factory) {
                if (typeof define === 'function' && define.amd) {
                    define(['jquery', 'underscore'], factory);
                } else if (typeof exports === 'object' && typeof module !== 'undefined') {
                    module.exports = factory(require('jquery'), require('underscore'));
                } else if (typeof exports === 'object') {
                    exports.NeExample = factory();
                } else {
                    root.NeExample = factory(root.jQuery, root._);
                }
            })(this, function ($, _) {
                var MyPlugin = {
                    init: function () {
                        console.log('初始化', $, _);
                    }
                };
                return MyPlugin;
            });

        


        模板使用注意事项：
        （1）浏览器直接引入时，必须先加载 jQuery、Underscore，再引入本插件；
        （2）RequireJS 环境会自动加载 jquery、underscore 模块，未配置模块路径会报错；
        （3）Node 项目需提前安装依赖：npm install jquery underscore；
        （4）插件内部代码可直接使用 $（jQuery）、_（Underscore）。

            })(this, function () {

                //~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
                // 1. 定义构造函数（核心：原型链的基础）
                //~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
                function MyPlugin(element, options) {
                    var me = this;
                    // 用户不使用 new 进行实例化时
                    if(! (this instanceof MyPlugin)) {
                        return new MyPlugin(element, options); // 自动使用 new 进行实例并返回
                    }
                    // 用户有使用 new 进行实例化时
                    if(arguments.length > 0) {
                        if(typeof element != 'undefined' && typeof options != 'undefined') {
                            me.init(element, options); // 初始化
                        }
                    }
                };



                //~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
                // 2. 原型链方法（核心：挂载到构造函数的 prototype 上）
                //~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
                // 初始化
                MyPlugin.prototype.init = function (element, options) {
                    var me = this;
                    // console.log('初始化');
                    var defaults = {
                        intervals: 100,
                        backgroundColor: 'beige',
                        color: '#000',
                        fontSize: 16
                    }
                    var settings = utils.merge(true, {}, defaults, options || {});
                    me.element = element;
                    me.defaults = defaults;
                    me.settings = settings;
                    me.changeColor(); // 改变颜色
                };

                // 改变颜色
                MyPlugin.prototype.changeColor = function () {
                    var me = this;
                    var bgColor = me.settings.backgroundColor,
                        fontColor = me.settings.color,
                        fontSize = me.settings.fontSize;
                    var selector = me.element.toString().replace(/(\#|\.)/g, '');
                    var elDom = document.getElementById(selector);
                    if(elDom != null) {
                        setTimeout(function() {
                            elDom.style.backgroundColor = bgColor;
                            elDom.style.color = fontColor;
                            elDom.style.fontSize = fontSize + 'px';
                        }, me.settings.intervals);
                    }
                };



                //~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
                // 3、私有函数：供内部调用
                //~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
                var helpers = {
                    _fn1: function () {
                        
                    },
                    _fn2: function () {
                        
                    }
                };


                //~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
                // 4、工具库：通用工具
                //~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
                var utils = {
                    
                    /**
                    * 原生JS对象合并
                    * 兼容IE6+，支持深浅合并，不污染原对象
                    * @param {Boolean} [deep] 可选，是否开启深拷贝；true深合并，false浅合并，不传默认false
                    * @param {Object} target 必传，合并基准目标对象
                    * @param {Object} source1 必传，待合并源对象1
                    * @param {Object} [source2,...sourceN] 可选，多个待合并源对象，支持无限个
                    * @returns {Object} 返回全新合并后的对象
                    * @usage
                    * 浅合并（第一层覆盖，嵌套对象直接替换）
                    * var res1 = utils.merge(false, {a:1}, {b:2}, {a:99});
                    * 深合并（嵌套对象递归合并，数组/日期/正则直接覆盖）
                    * var defaults = {name:"默认",info:{age:18}};
                    * var userOpts = {name:"张三",info:{sex:"男"}};
                    * var res2 = utils.merge(true, {}, defaults, userOpts);
                    */
                    merge: function () {
                        var deep = false;
                        var args = Array.prototype.slice.call(arguments);
                        // 判断第一个参数是否为布尔值，代表是否深拷贝
                        if (typeof args[0] === 'boolean') {
                            deep = args.shift();
                        }
                        // 没有传入需要合并的对象，直接返回空对象
                        if (args.length === 0) {
                            return {};
                        }
                        // 取出目标对象，手动浅拷贝一层，不污染原始入参
                        var target = args.shift();
                        var copyTarget = {};
                        for (var k in target) {
                            if (Object.prototype.hasOwnProperty.call(target, k)) {
                                copyTarget[k] = target[k];
                            }
                        }
                        target = copyTarget;
                        // 循环所有待合并源对象
                        for (var i = 0; i < args.length; i++) {
                            var source = args[i];
                            // 跳过null、undefined、基础类型
                            if (!source || typeof source !== 'object') {
                                continue;
                            }
                            // 遍历源对象自有属性
                            for (var key in source) {
                                if (!Object.prototype.hasOwnProperty.call(source, key)) {
                                    continue;
                                }
                                var srcVal = source[key];
                                var tarVal = target[key];
                                // 判断是否是纯普通对象：排除数组、Date、RegExp、null
                                var isPlainObj = srcVal !== null
                                    && typeof srcVal === 'object'
                                    && !Array.isArray(srcVal)
                                    && !(srcVal instanceof Date)
                                    && !(srcVal instanceof RegExp);
                                // 深合并逻辑：仅普通对象递归
                                if (deep && isPlainObj) {
                                    target[key] = this.merge(true, {}, tarVal || {}, srcVal);
                                } else {
                                    // 数组、日期、正则、基础类型直接覆盖
                                    target[key] = srcVal;
                                }
                            }
                        }
                        return target;
                    }
                };



                //~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
                // 5、暴露公共方法，供用户调用
                //~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
                return MyPlugin; // 返回对象：暴露构造函数（而非普通对象）
            });

 封装方法2：暴露构造函数 (调用原型链) 

        
 ———————————————————————————————————————————————————————————————————————————————————— 

        


            ;(function (root, factory) {
                if (typeof define === 'function' && define.amd) { // AMD规范
                    define(['jquery', 'underscore'], factory);
                } else if (typeof exports === 'object') { // Node, CommonJS规范
                    module.exports = factory(require('jquery'), require('underscore'));
                } else { // 浏览器全局变量(root 即 window)
                    root.NeExample = factory(root.jQuery, root._);
                }
            })(this, function ($, _) {
                // 1. 定义构造函数（核心：原型链的基础）
                function MyPlugin(options) {
                    // 构造函数初始化逻辑（比如接收配置参数）
                    this.options = options || {};
                    this.init(); // 初始化方法
                }

                // 2. 私有函数（仅插件内部可访问）
                function a() {
                    console.log('私有函数a，外部无法调用');
                }

                // 3. 原型链方法（核心：挂载到构造函数的 prototype 上）
                MyPlugin.prototype = {
                    // 重置构造函数指向（避免原型重写后 constructor 丢失）
                    constructor: MyPlugin,
                    
                    // 初始化方法（原型方法）
                    init: function () {
                        a(); // 内部可调用私有函数
                        console.log('初始化完成，配置：', this.options);
                    },
                    
                    // 原型方法 b（对应原插件的 b 函数）
                    protoB: function () {
                        console.log('原型方法b执行');
                        return this; // 支持链式调用
                    },
                    
                    // 原型方法 c（对应原插件的 c 函数）
                    protoC: function () {
                        console.log('原型方法c执行，依赖jQuery：', $);
                        console.log('依赖underscore：', _);
                        return this;
                    }
                };

                // 4. 暴露构造函数（而非普通对象）
                return MyPlugin;
            });

        


        
 前端调用方式：

        // 实例化构造函数，调用原型方法
        // 使用 new 创建实例并调用原型方法（支持链式调用）
        var example = new NeExample({
            key: '百度地图key',
            zoom: 10
        });
        example.protoB(); // 调用原型方法B
        example.protoC(); // 调用原型方法C。依赖jQuery/underscore
        example.protoB().protoC(); // 链式调用

            function createNewInstace(element, config){
                // 说明：bind()就是将函数绑定到某个对象上。 例如：f.bind(obj)，实际上可以理解为 obj.f()
                // 实例化一个对像
                var context = new NDate(element, config); // context 是一个实例化的对象，只能当对象使用，不能当函数使用。
                // 创建请求函数
                var instance = NDate.prototype.init.bind(context); // instance 是一个函数，由 bind 返回的一个新函数，可以调用 instance()。
                // 将 对象原型链 prototype 对象中的方法添加到 instance 函数对象中
                // 为了实现能够将 instance 函数作为对象使用，我们就要将 对象原型链 prototype 对象中的方法添加给 instance 。毕竟函数也一个对象，也能够添加方法。
                Object.keys(NDate.prototype).forEach(function(item){
                    instance[item] = NDate.prototype[item].bind(context);
                });
                // 为 instance 函数对象添加属性
                Object.keys(context).forEach(function(item){
                    instance[item] = context[item];
                })
                return instance;
            }
