Mock.js 同一个URL多action 解决方案（适配 .ashx / .net单入口接口）

场景说明：后端只有一个接口地址例如 /handler.ashx，依靠请求参数里面 action 区分不同业务；
不修改前端url，不需要本地加临时query参数，上线无需删除代码。

原理

Mock.mock 第二个参数method，第三个参数传入函数，函数接收options对象，可以读取请求body/url查询参数，内部根据action分支返回不同模拟数据。

完整mock注册代码示例（POST JSON，action在body里面）

// 模拟 .ashx 单入口，同一个url，不同action返回不同mock数据
var mockActionMap = {
    getStudentList: function () {
        return Mock.mock({
            code: "0",
            message: "success",
            data: {
                "records|4": [
                    {
                        "xuesheng_bh": "@id()",
                        "xuesheng_mc": "@cname()"
                    }
                ]
            }
        });
    },
    getActivityList: function () {
        return Mock.mock({
            code: "0",
            message: "success",
            data: {
                "records|2": [
                    {
                        "huodong_bh": "@id()",
                        "huodong_mc": "@ctitle(8,20)"
                    }
                ]
            }
        });
    },
    saveData: function () {
        return {
            code: "0",
            message: "保存成功",
            data: {}
        };
    }
};

// url使用正则匹配，防止query干扰匹配
Mock.mock(/^\/handler\.ashx/, "post", function (options) {
    // options.body 是post请求原始字符串
    var reqBody = JSON.parse(options.body);
    var action = reqBody.action;

    var handlerFn = mockActionMap[action];
    if (typeof handlerFn === "function") {
        return handlerFn();
    }
    // 未知action返回错误
    return {
        code: "90000",
        message: "未知action：" + action,
        data: {}
    };
});

前端业务调用示例（url全部相同，仅data内action不同）

// 获取学生列表
$.ajax({
    url: "/handler.ashx",
    type: "POST",
    contentType: "application/json",
    data: JSON.stringify({
        action: "getStudentList"
    })
}).done(function (res) {
    console.log("学生列表结果：", res);
});

// 获取活动列表
$.ajax({
    url: "/handler.ashx",
    type: "POST",
    contentType: "application/json",
    data: JSON.stringify({
        action: "getActivityList"
    })
}).done(function (res) {
    console.log("活动列表结果：", res);
});

// 保存数据
$.ajax({
    url: "/handler.ashx",
    type: "POST",
    contentType: "application/json",
    data: JSON.stringify({
        action: "saveData"
    })
}).done(function (res) {
    console.log("保存结果：", res);
});

如果是GET请求，action放在url query参数的写法

Mock.mock(/^\/handler\.ashx/, "get", function (options) {
    // 解析url中的query参数
    var urlObj = new URL(options.url, location.origin);
    var action = urlObj.searchParams.get("action");

    var handlerFn = mockActionMap[action];
    if (typeof handlerFn === "function") {
        return handlerFn();
    }
    return {
        code: "90000",
        message: "未知action：" + action,
        data: {}
    };
});

方案对比小结

方案说明缺点
url拼接临时?action=xxx mock区分接口 上线必须删除拼接参数，极易漏改，不推荐
多次Mock.mock注册同一个url 多个mock注册相同url 后面注册覆盖前面，完全失效，不可用
mock回调函数读取action分支 url保持原样，读取请求体内action做分支 前端代码完全不用改动，本地mock、线上后端一套代码，强烈推荐
关键点提醒：
1. options.body拿到的是字符串，必须 JSON.parse()；
2. url尽量使用正则匹配，避免query参数导致匹配失败；
3. 此方案对应小程序那套mockMap思想；小程序wx.request不能被mockjs拦截，需要自己在request.js内部读取data.action分支。
